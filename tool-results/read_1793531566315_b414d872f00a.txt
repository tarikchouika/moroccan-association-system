    28→    28→function Counter({ to, suffix = "", duration = 1600 }: { to: number; suffix?: string; duration?: number }) {
    29→    29→  const ref = useRef<HTMLSpanElement>(null);
    30→    30→  const inView = useInView(ref, { once: true, margin: "-40px" });
    31→    31→  const [val, setVal] = useState(0);
    32→    32→
    33→    33→  useEffect(() => {
    34→    34→    if (!inView) return;
    35→    35→    let raf = 0;
    36→    36→    const start = performance.now();
    37→    37→    const tick = (now: number) => {
    38→    38→      const p = Math.min((now - start) / duration, 1);
    39→    39→      setVal(Math.round(to * (1 - Math.pow(1 - p, 3))));
    40→    40→      if (p < 1) raf = requestAnimationFrame(tick);
    41→    41→    };
    42→    42→    raf = requestAnimationFrame(tick);
    43→    43→    return () => cancelAnimationFrame(raf);
    44→    44→  }, [inView, to, duration]);
    45→    45→
    46→    46→  return (
    47→    47→    <span ref={ref} className="num font-black">
    48→    48→      {val.toLocaleString("en-US")}
    49→    49→      {suffix}
    50→    50→    </span>
    51→    51→  );
    52→    52→}

    57→    57→  viewport: { once: true, margin: "-60px" },
    58→    58→  transition: { duration: 0.55, ease: "easeOut" as const },
    59→    59→};
